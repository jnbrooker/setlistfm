function ols(xs,ys){var n=xs.length,sx=0,sy=0,sxx=0,sxy=0;

var scMode="raw";
function drawSC(){
  var W=760,H=380,L=56,R=18,T=18,Bm=48,iw=W-L-R,ih=H-T-Bm;
  var lx=SCAT.map(function(d){return Math.log(d.k);}),
      ly=SCAT.map(function(d){return Math.log(d.e);}),
      lp=SCAT.map(function(d){return Math.log(d.p);});
  var X,Y,xlab,ylab;
  if(scMode==="raw"){X=lx;Y=ly;xlab="biggest indoor room (log seats)";ylab="shows (log)";}
  else{ // Frisch-Waugh: residualise both on log catchment
    var rx=ols(lp,lx),ry=ols(lp,ly);
    X=lx.map(function(v,i){return v-(rx.a+rx.b*lp[i]);});
    Y=ly.map(function(v,i){return v-(ry.a+ry.b*lp[i]);});
    xlab="room size, catchment removed";ylab="shows, catchment removed";}
  var f=ols(X,Y);
  var xmin=Math.min.apply(null,X),xmax=Math.max.apply(null,X),
      ymin=Math.min.apply(null,Y),ymax=Math.max.apply(null,Y);
  var px=(xmax-xmin)*.06,py=(ymax-ymin)*.08;
  xmin-=px;xmax+=px;ymin-=py;ymax+=py;
  var sx=function(v){return L+(v-xmin)/(xmax-xmin)*iw;},
      sy=function(v){return T+ih-(v-ymin)/(ymax-ymin)*ih;};
  var ink=css("--ink"),slate=css("--slate"),rule=css("--rule"),
      pine=css("--pine"),ember=css("--ember");
  var s='<rect x="'+L+'" y="'+T+'" width="'+iw+'" height="'+ih+'" fill="none" stroke="'+rule+'"/>';
  SCAT.forEach(function(d,i){
    s+='<circle cx="'+sx(X[i]).toFixed(1)+'" cy="'+sy(Y[i]).toFixed(1)+
       '" r="4.5" fill="'+pine+'" fill-opacity=".5" stroke="'+pine+'" stroke-width=".8"><title>'+
       esc(d.c)+" — "+d.e.toLocaleString()+" shows, "+d.k.toLocaleString()+
       " seats, "+d.p+"m catchment</title></circle>";
  });
  s+='<line x1="'+sx(xmin).toFixed(1)+'" y1="'+sy(f.a+f.b*xmin).toFixed(1)+
